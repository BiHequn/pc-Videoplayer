cmake_minimum_required(VERSION 3.21)
project(native_player LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "native-player requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(NP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/Project1/Project1")
set(NP_DEPENDENCY_ROOT "" CACHE PATH "Installed x64-windows-static dependency directory")
if(NOT NP_DEPENDENCY_ROOT)
  foreach(candidate "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg_installed/x64-windows-static" "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg_installed/x64-windows-static/x64-windows-static")
    if(EXISTS "${candidate}/include/libavformat/avformat.h")
      set(NP_DEPENDENCY_ROOT "${candidate}")
      break()
    endif()
  endforeach()
endif()
if(NOT EXISTS "${NP_DEPENDENCY_ROOT}/include/libavformat/avformat.h")
  message(FATAL_ERROR "Install dependencies with setup-vcpkg.ps1 or set NP_DEPENDENCY_ROOT")
endif()
add_library(player_media STATIC
  ${NP_SOURCE}/media/ffmpeg-decoder.cpp ${NP_SOURCE}/media/ffmpeg-runtime.cpp
  ${NP_SOURCE}/media/ffmpeg-video-decoder.cpp ${NP_SOURCE}/media/mf-video-decoder.cpp
  ${NP_SOURCE}/media/media-packet-queue.cpp ${NP_SOURCE}/media/video-frame-queue.cpp
  ${NP_SOURCE}/media/video-decoder-factory.cpp ${NP_SOURCE}/media/vod-downloader.cpp)
target_include_directories(player_media PUBLIC ${NP_SOURCE} "${NP_DEPENDENCY_ROOT}/include")
target_compile_definitions(player_media PUBLIC NOMINMAX UNICODE _UNICODE NP_ENABLE_FFMPEG_VIDEO_DECODER=1 NP_ENABLE_MF_VIDEO_DECODER=1)
target_compile_options(player_media PUBLIC /utf-8)
foreach(lib avformat avcodec swscale swresample avutil)
  target_link_libraries(player_media PUBLIC "${NP_DEPENDENCY_ROOT}/$<IF:$<CONFIG:Debug>,debug/lib,lib>/${lib}.lib")
endforeach()
target_link_libraries(player_media PUBLIC d3d11 dxgi d3dcompiler secur32 ws2_32 mfuuid ole32 strmiids user32 bcrypt winhttp shlwapi comctl32 comdlg32 winmm mfplat mf)
add_library(player_session STATIC ${NP_SOURCE}/app/playback-session.cpp)
target_link_libraries(player_session PUBLIC player_media)
set(IMGUI "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_compile_definitions(imgui PUBLIC NOMINMAX UNICODE _UNICODE)
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
add_executable(native-player
 ${NP_SOURCE}/app/main.cpp ${NP_SOURCE}/app/app-runtime.cpp ${NP_SOURCE}/app/source-selection-dialog.cpp
 ${NP_SOURCE}/app/desktop-player.cpp ${NP_SOURCE}/render/desktop-renderer.cpp
 ${NP_SOURCE}/platform/pip-window.cpp ${NP_SOURCE}/platform/parent-watch.cpp ${NP_SOURCE}/platform/player-channel.cpp
 ${NP_SOURCE}/render/d3d11-renderer.cpp ${NP_SOURCE}/render/d3d11-shader.cpp ${NP_SOURCE}/render/d3d11-video-texture.cpp)
target_link_libraries(native-player PRIVATE player_session imgui windowscodecs)
set_target_properties(native-player PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_options(native-player PRIVATE /ENTRY:mainCRTStartup)
set_target_properties(native-player PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/desktop")
include(CTest)
if(BUILD_TESTING)
  add_executable(make-test-media tests/make-test-media.cpp)
  target_link_libraries(make-test-media PRIVATE player_media)
  add_executable(playback-session-test tests/playback-session-test.cpp)
  target_link_libraries(playback-session-test PRIVATE player_session)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/media-controls-test.cpp")
    add_executable(media-controls-test tests/media-controls-test.cpp)
    target_link_libraries(media-controls-test PRIVATE player_media)
    add_test(NAME media-controls COMMAND media-controls-test "${CMAKE_CURRENT_BINARY_DIR}/fixtures/audiovisual.avi" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/silent.avi")
    set_tests_properties(media-controls PROPERTIES FIXTURES_REQUIRED media TIMEOUT 45)
  endif()
  add_test(NAME generate-fixtures COMMAND make-test-media "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  set_tests_properties(generate-fixtures PROPERTIES FIXTURES_SETUP media)
  add_test(NAME playback-session COMMAND playback-session-test "${CMAKE_CURRENT_BINARY_DIR}/fixtures/audiovisual.avi")
  set_tests_properties(playback-session PROPERTIES FIXTURES_REQUIRED media TIMEOUT 45)
  add_test(NAME desktop-smoke COMMAND native-player --source "${CMAKE_CURRENT_BINARY_DIR}/fixtures/audiovisual.avi" --decoder ffmpeg --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/screenshots")
  set_tests_properties(desktop-smoke PROPERTIES FIXTURES_REQUIRED media TIMEOUT 30 RUN_SERIAL TRUE)
endif()
